cmake_minimum_required(VERSION 3.15)
project(myfft CXX)
add_library(myfft STATIC fft_wrapper.cpp)
target_include_directories(myfft PRIVATE ${KFR_ROOT}/include)
target_compile_options(myfft PRIVATE -ffp-contract=fast)
target_compile_definitions(myfft PRIVATE NDEBUG)
set_target_properties(myfft PROPERTIES PUBLIC_HEADER fft_wrapper.h)
install(TARGETS myfft ARCHIVE DESTINATION lib PUBLIC_HEADER DESTINATION include)
file(GLOB KFR_LIBS ${KFR_ROOT}/lib/libkfr_dft*.a ${KFR_ROOT}/lib/libkfr_dsp*.a)
install(FILES ${KFR_LIBS} DESTINATION lib)
